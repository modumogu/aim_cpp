# ============================================================================
#  官方仿真 robomini_sim 专用参数
# ============================================================================
#  用途：在 tianze-dev/robomini_sim 里跑题4 时使用
#  用法： aim.exe --config config_robomini.txt --video <录制的画面>
#
#  和 config.txt（真机）的区别只有三块：相机内参、画面尺度、弹道来源。
#  算法本身完全一样。
# ============================================================================


# ---------------------------------------------------------------------------
# 相机内参 —— 官方仿真白给的，不用标定
# ---------------------------------------------------------------------------
# SDF 里写的是 horizontal_fov = 1.047198 rad（60°）、1280x720
#   fx = fy = (1280/2) / tan(60°/2) = 640 / 0.57735 = 1108.5
#   cx = 640, cy = 360（画面正中心）
# Gazebo 的相机是理想针孔，没有畸变，所以 k1/k2/p1/p2/k3 全填 0。
#
# 注意：如果改了 image 的宽高（比如改成 1280x1024），cx/cy 要跟着改一半值，
#       但 fx/fy 不变（焦距由 horizontal_fov 和宽度决定）。
fx = 1108.5
fy = 1108.5
cx = 640
cy = 360
k1 = 0
k2 = 0
p1 = 0
p2 = 0
k3 = 0


# ---------------------------------------------------------------------------
# 画面尺度
# ---------------------------------------------------------------------------
# 相机到转盘中心的距离约 1954mm（发射架 y=-2.07，靶面 y=-0.116）
#   px_per_mm = 1108.5 / 1954 = 0.567
# 转盘半径 350mm -> 约 199 像素；靶盘 195mm -> 约 111 像素
#
# 官方仿真的灯条只有约 23x11 像素（真机是 35x17），所以形状阈值要放宽一点。
px_per_mm = 0.567

# 轮心默认用画面正中心。前提是相机正对转盘中心，
# 如果装偏了，跑一次看画面里转盘中心落在哪，改成实际像素。
wheel_x = -1
wheel_y = -1

# 自动估计半径：从检出的灯条反算，比手算 px_per_mm 更准
auto_rpx = 1
roi_scale = 1.6


# ---------------------------------------------------------------------------
# 灯条形状 —— 仿真里目标更小，阈值比真机宽松
# ---------------------------------------------------------------------------
# 灯条 23x11 px，面积约 250 px²（真机是 600）
min_area = 20
max_area = 2500

# 长宽比：理论 41/20 = 2.05，渲染带抗锯齿后约 1.8，取 1.2 留余量
min_aspect = 1.2

# 填充度：23x11 的矩形四边各蚀掉半像素约 0.87，取 0.65 留余量
min_extent = 0.65

# 色相一致性（度）
max_hue_std = 6.0


# ---------------------------------------------------------------------------
# 槽位归组
# ---------------------------------------------------------------------------
group_gap_deg = 25
radius_tol = 0.45


# ---------------------------------------------------------------------------
# 拟合与预测
# ---------------------------------------------------------------------------
fit_window_s = 1.5
min_fit_samples = 20
omega_min = 1.884
omega_max = 2.000
omega_step = 0.004
max_fit_rms = 0.05
min_osc_amp = 0.05
ab_tol = 0.20

lead_s = 0.10
max_pred_age = 0.25
history_reset_age = 1.0
model_gate = 1.5


# ---------------------------------------------------------------------------
# 相机安装姿态
# ---------------------------------------------------------------------------
# 真机方案是"相机固定在基座、水平略抬头"。仿真里把 sensor 从 pitch_link
# 挪到 base_link 后，同样要填这个抬头角。
#
# 抬头角 = atan((转盘中心高 - 相机高) / 水平距离)
#   相机 0.60m、转盘中心 0.928m、距离 1.954m
#   -> atan(0.328 / 1.954) = 9.53°
#
# 如果仿真里相机改成完全水平（pitch=0），这里填 0。
camera_pitch_deg = 9.53
camera_yaw_deg = 0


# ---------------------------------------------------------------------------
# 靶的几何
# ---------------------------------------------------------------------------
# 官方 SDF：4 条灯条质心距盘心 47.5mm，正十字分布
bar_offset_mm = 47.5


# ---------------------------------------------------------------------------
# 弹道
# ---------------------------------------------------------------------------
# 仿真的枪口速度由 world 里 default_fire_speed 决定，默认 12 m/s。
# launch 时用 fire_speed:= 改的话，这里也要跟着改。
ball_speed = 12

# 枪管轴比相机光轴低多少（mm）。按真机方案写 75，仿真里如果是同轴就填 0。
muzzle_offset_mm = 75

# 视觉+串口+云台响应的延迟（秒），不含飞行时间。
# 仿真里视觉处理比真机快，可以先填小一点，实测再调。
pipeline_delay = 0.08

# 实弹标定的经验修正，仿真里先填 0
pitch_bias_deg = 0


# ---------------------------------------------------------------------------
# 颜色
# ---------------------------------------------------------------------------
# 颜色不在这里配，用命令行 --color red|blue 切换（默认 red）。
# 仿真 launch 的 side:=blue / side:=red 要和这个对应。
