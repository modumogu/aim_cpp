#
#  语法：
#     键 = 值          # 井号后面整行都是注释，可以写中文
#
#  优先级： 命令行参数  >  本文件  >  代码里的默认值
#           临时试一个值用命令行，例如：  aim.exe --gate 0.5
#
#  文件请存成 UTF-8。键名和值必须是英文/数字，注释随便写。

# ---------------------------------------------------------------------------
# 相机
# ---------------------------------------------------------------------------
# 曝光：DirectShow 的 log2 刻度。-8 约 3.9ms，-7 约 7.8ms，-6 约 15.6ms。
# 数值越大画面越亮。现场太暗就调大，拖影太明显就调小。
exposure = -8

# 手动色温（开尔文）。自动白平衡必须关，否则纯红会被拍成黄色。
# 现场灯光偏黄就往小调（3200），偏蓝就往大调（5600）。
wb = 4600

# 固定增益。一定要锁死，自动增益会把噪声放大成满屏碎块。
gain = 0


# ---------------------------------------------------------------------------
# 转盘在画面里的位置（现场必须标定）
# ---------------------------------------------------------------------------
# 相机装好后，把转盘放在正前方，看画面里转盘圆心落在哪个像素，填这两个数。
# 填 -1 表示用画面正中心（只适合仿真）。
wheel_x = -1
wheel_y = -1

# 每毫米多少像素 = 焦距 / 距离。标定得到 fx=1700、距离 1950mm 时为 0.8720。
# 换镜头或改安装距离都要重算。
px_per_mm = 0.8720

# 自动估计转盘半径：1 = 用检出的灯条自动算 R，0 = 只用下面的 px_per_mm。
# 建议开着：换相机、换距离、拿测试视频跑都不用改 px_per_mm。
auto_rpx = 1

# 搜索范围：只在这个半径倍数内找灯条。顶灯、墙面反光通常在圆外。
# 1.6 倍 = 约 488 像素；设成 0 关闭 ROI（全部画面都找）。
roi_scale = 1.6


# ---------------------------------------------------------------------------
# 灯条形状过滤
# ---------------------------------------------------------------------------
# 官方灯条 41x20mm，在 1950mm 下约 36x17 像素，带光晕会涨到 45x25。
# 面积区间（平方像素）。太小是噪点，太大是光斑或整片红色。
min_area = 20
max_area = 2500

# 长宽比下限。实测真实灯条 1.73~1.80（光晕把宽度撑大），取 1.2 留余量。
min_aspect = 1.2

# 填充度 = 轮廓面积 / 最小外接矩形面积。实测 0.873~0.915。
# 眩光和不规则反光通常低于 0.6，取 0.70 分开两边。
# 真机保守取 0.65；若现场过曝导致漏检，可降到 0.50（Linux 侧验证过）
min_extent = 0.65

# 色相一致性的允许标准差（度）。真实 LED 单一色相 1~3 度；
# 白色高光边的紫边/红边超过 10 度。取 6 度。
max_hue_std = 12.0


# ---------------------------------------------------------------------------
# 槽位归组
# ---------------------------------------------------------------------------
# 角度聚类阈值（度）。必须落在 15.5 ~ 56.5 之间：
#   组内最大间隔 = 2*atan(47.5/350) = 15.46 度（径向灯漏检时）
#   组间最小间隔 = 72 - 15.46 = 56.54 度
group_gap_deg = 25

# 半径容差。灯条离轮心的距离允许偏离 R 的比例，0.45 即 ±45%。
radius_tol = 0.45


# ---------------------------------------------------------------------------
# 拟合 θ(t)
# ---------------------------------------------------------------------------
# 只用最近多少秒的数据。太短噪声大，太长跟不上变速。
fit_window_s = 1.5

# 至少多少个采样点才允许拟合。
min_fit_samples = 20
# 最小拟合时间跨度（秒）。只有样本数不够：高帧率下 20 帧可能只覆盖 0.16 秒，
# 正弦拟合会欠定。60fps 时 20 帧=0.33s，129fps 时只有 0.16s。
min_fit_span_s = 0.30

# 运行时帧率看门狗（只对相机模式生效）。低于 fps_warn_low 持续 2 秒就报警，
# 恢复到 fps_warn_high 以上重新布防。现场掉帧多是中途插了 USB 设备抢带宽。
fps_warn_low = 50
fps_warn_high = 65


# ω 搜索范围（弧度/秒）。手册 5.5.2 规定 1.884 ~ 2.000，步长 0.004。
omega_min = 1.884
omega_max = 2.000
omega_step = 0.004

# 拟合残差上限（弧度）。超过就 HOLD，不开火。
# 0.05 弧度约 3 度，靶盘张角才 5.7 度，残差超过就没意义了。
max_fit_rms = 0.05

# 判定"确实在正弦摆动"的最小振幅（弧度/秒）。
# 低于这个值就当匀速运动，不做 a+b 约束检查（小机关和固定靶不适用）。
min_osc_amp = 0.05

# 手册约束 a+b=2.090 的允许偏差（弧度）。
# 实测本管线 P99 约 0.104，取 0.20 不误杀。
ab_tol = 0.20


# ---------------------------------------------------------------------------
# 预测与安全门限
# ---------------------------------------------------------------------------
# 预测提前量（秒）。覆盖视觉处理 + 串口传输 + 云台响应。
lead_s = 0.10

# 数据年龄上限（秒）。超过就 HOLD：大机关 120 度/秒，250ms 位置已偏 30 度。
max_pred_age = 0.25

# 历史清空阈值（秒）。丢帧超过这个时间就清空重来（转盘可能换模式或被复位）。
history_reset_age = 1.0

# 模型门限（度）。用上一帧模型外推本帧相位，偏差超过这个值就丢弃该帧，
# 防止单帧误检把累加的 θ 永久带偏。实测逐帧误差小于 0.2 度，取 1.5 有 7 倍余量。
model_gate = 1.5

# ---------------------------------------------------------------------------
# 相机内参（棋盘格标定结果，py_sim\calibrate.py 输出）
# ---------------------------------------------------------------------------
# 换相机、重装镜头都要重新标定并更新这里。
fx = 1700.37
fy = 1701.76
cx = 647.93
cy = 490.34
k1 = -0.1262
k2 = 0.5895
p1 = -0.0011
p2 = 0.0009
k3 = -1.0703


# ---------------------------------------------------------------------------
# 靶的几何（官方仿真 energy_rune/model.sdf）
# ---------------------------------------------------------------------------
# 4 条灯条质心距靶盘中心 47.5mm，成十字分布（径向 ±47.5、切向 ±47.5）。
# 真机如果和仿真不一致，用卡尺量了改这里。
bar_offset_mm = 47.5


# ---------------------------------------------------------------------------
# 弹道
# ---------------------------------------------------------------------------
# 乒乓球初速（m/s）。规则没规定，由摩擦轮决定；仿真默认 12。
# 实测方法：对着固定靶打十几发，量弹着点相对瞄准点的偏差反推。
ball_speed = 12

# 枪管轴比相机光轴低多少（mm）。相机装在枪管上方，所以从枪口看目标更高。
# 官方 SDF：相机在 pitch_link 上方 75mm，枪管在下方。
muzzle_offset_mm = 75

# 视觉 + 串口 + 云台响应的延迟（秒），不含弹丸飞行时间。
# 实际提前量 = 这个值 + 飞行时间，程序自动算。
pipeline_delay = 0.10

# 实弹打靶后的经验俯仰修正（度）。打低了就往正调，打高了往负调。
pitch_bias_deg = 0

# ---------------------------------------------------------------------------
# 相机安装姿态（真机必须填，这一项错了 pitch 会跟着错）
# ---------------------------------------------------------------------------
# 相机固定在基座上、略微抬头。抬头角按下面的式子算：
#     抬头角 = atan((转盘中心高 - 相机高) / 水平距离)
#   例：相机装 0.60m、转盘中心 0.928m、射距 1.954m
#       -> atan(0.328 / 1.954) = 9.53 度
#
# 不填（或填 0 = 完全水平）的后果：最上面那块靶会出画面（0.61m 高时差 4.6 度）。
# 填错（比如填 15 但实际是 9）的后果：竖直方向被过度拉伸，靶的角度会有系统偏差。
#
# 相机正对转盘中心竖直方向时 cam_yaw 填 0。
camera_pitch_deg = 0
camera_yaw_deg = 0

# 重力加速度，只在做弹道计算时用，一般不用改
g_acc = 9.8

# ---------------------------------------------------------------------------
# 亮度 / 饱和度自适应（压力测试后新增，正常光照下可以不改）
# ---------------------------------------------------------------------------
# 画面整体偏暗时自动补增益（软件自动曝光），避免 V 阈值把灯条一起挡掉。
# 只有 ROI 内亮度的 P99 低于 adapt_gain_below 才补，正常光照不会触发。
adapt_sv = 1
adapt_gain_below = 110
adapt_max_gain = 4.0

# 饱和度自适应：只有"画面里最亮的那些像素"本身就偏淡（S<120）时才放宽 S 下限，
# 用于应对褪色/白平衡失配。暖光下灯条仍然饱和，不会触发，避免背景混入。
adapt_ratio = 0.45
adapt_min = 45
